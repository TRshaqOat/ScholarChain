   146	      failPuts: AtomicInteger = AtomicInteger(0),
   147	      calls: AtomicInteger = AtomicInteger()
   148	  ) =
   149	    val items = scala.collection.mutable.HashMap.empty[(String, String), java.util.Map[String, Sdk]]
   150	    def key(m: java.util.Map[String, Sdk]) = (m.get("recordKey").s, m.get("recordType").s)

   152	      case ("getItem", Seq(r: GetItemRequest)) =>
   153	        val k = (r.key.get("recordKey").s, r.key.get("recordType").s)
   154	        items
   155	          .synchronized(items.get(k))
   156	          .fold(GetItemResponse.builder().build())(i => GetItemResponse.builder().item(i).build())
   157	      case ("putItem", Seq(r: PutItemRequest)) =>
   158	        calls.incrementAndGet()
   159	        if failPuts.getAndUpdate(n => math.max(0, n - 1)) > 0 then
   160	          throw throttle("ProvisionedThroughputExceededException")

   162	          val stored = items.get(key(r.item))
   163	          val now = r.expressionAttributeValues.asScala.get(":now").map(_.n.toLong)
   164	          val from = r.expressionAttributeValues.asScala.get(":from").map(_.s)
   165	          val ok = (now, from) match
   166	            case (Some(t), _) => stored.forall(_.get("expiresAt").n.toLong <= t)
   167	            case (_, Some(s)) => stored.exists(_.get("state").s == s)
   168	            case _            => true
   169	          if !ok then
   170	            val b =
   171	              ConditionalCheckFailedException.builder().message("The conditional request failed")
   172	            throw stored.fold(b)(b.item).build()

   175	        }

   364	  final class Client(val port: Int):
   365	    def send(
   366	        method: String,
   367	        path: String,
   368	        body: Option[HttpRequest.BodyPublisher] = None,
   369	        headers: Seq[(String, String)] = JsonHeaders
   370	    ): Reply =
   371	      val b = HttpRequest
   372	        .newBuilder(URI.create(s"http://127.0.0.1:$port$path"))
   373	        .timeout(Duration.ofSeconds(30)) // hang guard only; a healthy service answers at once
   374	        .method(method, body.getOrElse(HttpRequest.BodyPublishers.noBody()))
   375	      headers.foreach((k, v) => b.header(k, v))
   376	      val r = http.send(b.build(), HttpResponse.BodyHandlers.ofString())
   377	      val hs = Map.from(
   378	        Seq("X-Request-Id", "X-Trace-Id", "Idempotent-Replay", "Retry-After", "Content-Type")
   379	          .flatMap(h => Option(r.headers.firstValue(h).orElse(null)).map(h -> _))
   380	      )
   381	      val parsed =
   382	        if r.body.isEmpty then ujson.Null
   383	        else scala.util.Try(ujson.read(r.body)).getOrElse(ujson.Str(r.body))
   384	      Reply(r.statusCode, parsed, hs)
   385	    def get(path: String, headers: Seq[(String, String)] = Seq("X-Api-Key" -> ApiKey)): Reply =
   386	      send("GET", path, None, headers)
   387	    def post(path: String, body: String, headers: Seq[(String, String)] = Seq()): Reply =
   388	      send("POST", path, Some(HttpRequest.BodyPublishers.ofString(body)), JsonHeaders ++ headers)

   119	          l.edge.props("jointPublicationCount")
