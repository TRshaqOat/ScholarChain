   143	def worker(host, port, key, paths, offset, deadline, cap, level, wid, rows, lock):
   144	    headers = {"X-Api-Key": key, "X-Request-Id": "exp-d-c%d-w%d" % (level, wid)}
   145	    conn = None
   146	    i = offset
   147	    while time.monotonic() < deadline:
   148	        with lock:
   149	            if len(rows) >= cap:
   150	                return
   151	        kind, path = paths[i % len(paths)]
   152	        i += 1
   153	        start = utc_now()
   154	        t0 = time.perf_counter()
   155	        status, err = 0, ""
   156	        try:
   157	            if conn is None:
   158	                conn = http.client.HTTPConnection(host, port, timeout=REQUEST_TIMEOUT_S)
   159	            conn.request("GET", path, headers=headers)
   160	            resp = conn.getresponse()
   161	            resp.read()
   162	            status = resp.status
   163	            if (resp.getheader("Connection") or "").lower() == "close":
   164	                conn.close()
   165	                conn = None
   166	        except Exception as e:  # recorded as its type only, never with headers
   167	            err = type(e).__name__
   168	            if conn is not None:
   169	                conn.close()
   170	            conn = None
   171	        ms = (time.perf_counter() - t0) * 1000.0
   172	        with lock:
   173	            rows.append([level, wid, start, kind, path, status, "%.1f" % ms, err])
   174	    if conn is not None:
   175	        conn.close()

   178	def run_level(base, key, paths, level, seconds, out_dir):
   179	    u = urllib.parse.urlsplit(base)
   180	    rows, lock = [], threading.Lock()
   181	    deadline = time.monotonic() + seconds
   182	    started = utc_now()
   183	    ts = [threading.Thread(target=worker, args=(u.hostname, u.port or 80, key, paths,
   184	                                                w * (len(paths) // level or 1), deadline,
   185	                                                MAX_REQUESTS_PER_LEVEL, level, w, rows, lock))
   186	          for w in range(level)]
   187	    for t in ts:
   188	        t.start()
   189	    for t in ts:
   190	        t.join()
   191	    ended = utc_now()
   192	    with open(os.path.join(out_dir, "requests-c%d.csv" % level), "w", newline="") as f:
   193	        w = csv.writer(f)
   194	        w.writerow(["concurrency", "worker", "startUtc", "kind", "path", "status", "latencyMs", "error"])
   195	        w.writerows(rows)
   196	    lat = sorted(float(r[6]) for r in rows)
   197	    ok = sorted(float(r[6]) for r in rows if 200 <= r[5] < 300)
   198	    errors = sum(1 for r in rows if not 200 <= r[5] < 300)
   199	    by_status = {}
   200	    for r in rows:
   201	        by_status[str(r[5])] = by_status.get(str(r[5]), 0) + 1
   202	    s = {"concurrency": level, "startedAt": started, "endedAt": ended, "seconds": seconds,
   203	         "requests": len(rows), "throughputPerSecond": round(len(rows) / float(seconds), 2),
   204	         "errors": errors, "errorRate": round(errors / float(len(rows)), 4) if rows else None,
   205	         "statusCounts": by_status,
   206	         "latencyMsAll": {"p50": percentile(lat, 50), "p95": percentile(lat, 95), "p99": percentile(lat, 99)},
   207	         "latencyMs2xx": {"p50": percentile(ok, 50), "p95": percentile(ok, 95), "p99": percentile(ok, 99)},
   208	         "requestCapReached": len(rows) >= MAX_REQUESTS_PER_LEVEL}
   209	    print("    level c=%d requests=%d rps=%s errors=%d p50=%s p95=%s p99=%s" % (
   210	        level, s["requests"], s["throughputPerSecond"], errors, s["latencyMsAll"]["p50"],
   211	        s["latencyMsAll"]["p95"], s["latencyMsAll"]["p99"]))
   212	    return s
